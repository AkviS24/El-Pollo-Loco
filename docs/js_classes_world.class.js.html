/**
 * Repräsentiert die Spielwelt, in der Charakter, Gegner, Items und das HUD verwaltet werden.
 */
class World {

    /**
     * Hauptcharakter des Spiels
     * @type {Character}
     */
    character = new Character();

    /**
     * Aktuelles Level
     * @type {Level}
     */
    level;

    /**
     * Der Endboss des Levels
     * @type {Endboss}
     */
    endboss;

    /**
     * Zeichenkontext des Canvas
     * @type {CanvasRenderingContext2D}
     */
    ctx;

    /**
     * HTML-Canvas-Element
     * @type {HTMLCanvasElement}
     */
    canvas;

    /**
     * Tastatursteuerung
     * @type {Keyboard}
     */
    keyboard;

    /**
     * X-Verschiebung der Kamera
     * @type {number}
     */
    cameraX = 0;

    /**
     * Lebensanzeige von Pepe
     * @type {StatusBar}
     */
    healthBar = new StatusBar("health", 10, 0);

    /**
     * Münzanzeige
     * @type {StatusBar}
     */
    coinBar = new StatusBar("coin", 10, 42);

    /**
     * Flaschenanzeige
     * @type {StatusBar}
     */
    bottleBar = new StatusBar("bottle", 10, 84);

    /**
     * Lebensanzeige des Endbosses
     * @type {StatusBar}
     */
    endbossBar = new StatusBar("endboss", 505, 58);

    /**
     * Geworfene Flaschen
     * @type {ThrowableObject[]}
     */
    throwableObjects = [];

    /**
     * Aufsteigende Münz-Animationen
     * @type {CoinAnimation[]}
     */
    coinAnimations = [];

    /**
     * Preis einer Flasche in Münzen
     * @type {number}
     */
    bottlePrice = 5;

    /**
     * Zeitpunkt des letzten Flaschenkaufs
     * @type {number}
     */
    lastBuyTime = 0;

    /**
     * TRUE, sobald das Spiel gewonnen oder verloren ist
     * @type {boolean}
     */
    gameEnded = false;

    /**
     * ID des aktuellen Animation-Frames
     * @type {number}
     */
    animationFrame;

    /**
     * Erstellt eine neue Spielwelt und startet Zeichen- und Spiellogik.
     *
     * @param {HTMLCanvasElement} canvas - Canvas-Element, auf dem die Welt gezeichnet wird
     * @param {Keyboard} keyboard - Steuerung für das Spiel
     * @param {Level} level - Das zu spielende Level
     */
    constructor(canvas, keyboard, level) {
        this.ctx = canvas.getContext("2d");
        this.canvas = canvas;
        this.keyboard = keyboard;
        this.level = level;
        this.endboss = level.enemies.find((enemy) => enemy instanceof Endboss);
        this.setWorld();
        this.draw();
        this.run();
    }

    /** Verbindet Charakter und Gegner mit der Welt. */
    setWorld() {
        this.character.world = this;
        this.level.enemies.forEach((enemy) => enemy.world = this);
    }

    /** Startet die Spiellogik mit 60 Aktualisierungen pro Sekunde. */
    run() {
        setStoppableInterval(() => this.update(), 1000 / 60);
    }

    /** Führt alle Prüfungen eines Spiel-Ticks aus. */
    update() {
        this.checkEnemyCollisions();
        this.checkThrowObjects();
        this.checkBottleHits();
        this.checkCollectables();
        this.checkCoinsToBottles();
        this.removeFinishedObjects();
        this.updateStatusBars();
        this.checkGameEnd();
    }

    /**
     * Prüft Kollisionen zwischen Pepe und den Gegnern.
     * Springt Pepe von oben auf ein Huhn, wird es besiegt – sonst nimmt Pepe Schaden.
     */
    checkEnemyCollisions() {
        if (this.character.isDead()) return;
        this.level.enemies.forEach((enemy) => {
            if (enemy.isDead() || !this.character.isColliding(enemy)) return;
            if (this.isStomping(enemy)) this.stompEnemy(enemy);
            else this.character.hit(enemy.damage);
        });
    }

    /**
     * Prüft, ob Pepe von oben auf einen Gegner fällt.
     * Dafür müssen seine Füße vor dem letzten Fall-Schritt noch über dem Gegner gewesen sein –
     * wer seitlich in einen Gegner hineinspringt, nimmt Schaden.
     *
     * @param {Enemy} enemy - Der berührte Gegner
     * @returns {boolean} TRUE, wenn der Gegner zertreten wird
     */
    isStomping(enemy) {
        let pepe = this.character;
        let previousFeetY = pepe.y + pepe.height - pepe.offset.bottom + pepe.speedY + pepe.acceleration;
        let enemyTop = enemy.y + enemy.offset.top;
        return enemy.stompable && pepe.speedY < 0 && previousFeetY <= enemyTop + 15;
    }

    /**
     * Besiegt einen Gegner durch Draufspringen und lässt Pepe abprallen.
     *
     * @param {Enemy} enemy - Der besiegte Gegner
     */
    stompEnemy(enemy) {
        enemy.die();
        this.character.bounce();
        this.rewardCoin(enemy);
    }

    /**
     * Belohnt Pepe für einen besiegten Gegner mit einer Münze.
     *
     * @param {Enemy} enemy - Der besiegte Gegner
     */
    rewardCoin(enemy) {
        this.character.coins++;
        this.coinAnimations.push(new CoinAnimation(enemy.x, enemy.y - 40));
    }

    /** Wirft eine Flasche, wenn F gedrückt ist und die Wurfpause vorbei ist. */
    checkThrowObjects() {
        if (!this.keyboard.F || !this.character.canThrow()) return;
        this.character.lastThrowTime = Date.now();
        if (this.character.bottles > 0) this.throwBottle();
        else soundManager.play("empty");
    }

    /** Erstellt eine Flasche an Pepes Hand, die in seine Blickrichtung fliegt. */
    throwBottle() {
        let pepe = this.character;
        let x = pepe.otherDirection ? pepe.x - 10 : pepe.x + pepe.width - 50;
        this.throwableObjects.push(new ThrowableObject(x, pepe.y + 100, pepe.otherDirection));
        pepe.bottles--;
        pepe.wakeUp();
    }

    /** Prüft, ob geworfene Flaschen einen Gegner oder den Boden treffen. */
    checkBottleHits() {
        this.throwableObjects.forEach((bottle) => {
            if (bottle.isSplashing) return;
            let enemy = this.level.enemies.find((e) => !e.isDead() && bottle.isColliding(e));
            if (enemy) this.hitEnemyWithBottle(enemy, bottle);
            else if (!bottle.isAboveGround()) bottle.splash();
        });
    }

    /**
     * Verarbeitet einen Flaschentreffer auf einen Gegner.
     *
     * @param {Enemy} enemy - Der getroffene Gegner
     * @param {ThrowableObject} bottle - Die treffende Flasche
     */
    hitEnemyWithBottle(enemy, bottle) {
        bottle.splash();
        enemy.takeBottleHit();
        if (enemy.isDead() && enemy.stompable) this.rewardCoin(enemy);
    }

    /** Prüft, ob Pepe Münzen oder Flaschen einsammelt. */
    checkCollectables() {
        this.collect(this.level.coins, () => this.collectCoin());
        this.collect(this.level.bottles, () => this.collectBottle());
    }

    /**
     * Entfernt alle Objekte, die Pepe berührt, aus der Liste und ruft die Belohnung auf.
     *
     * @param {MovableObject[]} items - Liste der Sammelobjekte
     * @param {Function} onCollect - Wird pro eingesammeltem Objekt aufgerufen
     */
    collect(items, onCollect) {
        for (let i = items.length - 1; i >= 0; i--) {
            if (!this.character.isColliding(items[i])) continue;
            items[i].stopIntervals();
            items.splice(i, 1);
            onCollect();
        }
    }

    /** Fügt eine Münze hinzu und spielt den Sammelsound ab. */
    collectCoin() {
        this.character.coins++;
        soundManager.play("coin");
    }

    /** Fügt eine Flasche hinzu und spielt den Sammelsound ab. */
    collectBottle() {
        this.character.bottles++;
        soundManager.play("bottle");
    }

    /** Prüft, ob mit V eine Flasche gekauft werden soll (höchstens alle 400 ms). */
    checkCoinsToBottles() {
        if (!this.keyboard.V || this.character.isDead() || Date.now() - this.lastBuyTime < 400) return;
        this.lastBuyTime = Date.now();
        if (this.character.coins >= this.bottlePrice) this.buyBottle();
        else soundManager.play("empty");
    }

    /** Tauscht Münzen gegen eine Flasche. */
    buyBottle() {
        this.character.coins -= this.bottlePrice;
        this.collectBottle();
    }

    /** Entfernt besiegte Gegner, zerplatzte Flaschen und beendete Animationen. */
    removeFinishedObjects() {
        this.level.enemies = this.level.enemies.filter((enemy) => !enemy.removable);
        this.throwableObjects = this.throwableObjects.filter((bottle) => !bottle.removable);
        this.coinAnimations = this.coinAnimations.filter((coin) => !coin.removable);
    }

    /** Aktualisiert alle Statusleisten anhand der aktuellen Werte. */
    updateStatusBars() {
        this.healthBar.setPercentage(this.character.energy);
        this.coinBar.setPercentage(this.character.coins * 10);
        this.coinBar.count = this.character.coins;
        this.bottleBar.setPercentage(this.character.bottles * 10);
        this.bottleBar.count = this.character.bottles;
        this.endbossBar.setPercentage(this.endboss.energy);
    }

    /** Prüft, ob Pepe gestorben ist oder der Endboss besiegt wurde. */
    checkGameEnd() {
        if (this.gameEnded) return;
        if (this.character.isDead()) this.endGame(false);
        else if (this.endboss.isDead()) this.endGame(true);
    }

    /**
     * Beendet das Spiel nach einer kurzen Pause, damit die Todesanimation sichtbar bleibt.
     *
     * @param {boolean} won - TRUE bei Sieg, FALSE bei Niederlage
     */
    endGame(won) {
        this.gameEnded = true;
        soundManager.pause("walk");
        soundManager.pause("music");
        scheduleGameEnd(won);
    }

    /**
     * Zeichnet die Welt und das HUD und fordert den nächsten Frame an.
     */
    draw() {
        this.ctx.clearRect(0, 0, this.canvas.width, this.canvas.height);
        this.ctx.translate(this.cameraX, 0);
        this.drawWorldObjects();
        this.ctx.translate(-this.cameraX, 0);
        this.drawHud();
        this.animationFrame = requestAnimationFrame(() => this.draw());
    }

    /** Zeichnet alle Objekte, die sich mit der Kamera bewegen. */
    drawWorldObjects() {
        this.addObjectsToMap(this.level.backgroundObjects);
        this.addObjectsToMap(this.level.clouds);
        this.addObjectsToMap(this.level.bottles);
        this.addObjectsToMap(this.level.coins);
        this.addObjectsToMap(this.level.enemies);
        this.addToMap(this.character);
        this.addObjectsToMap(this.throwableObjects);
        this.addObjectsToMap(this.coinAnimations);
    }

    /** Zeichnet die fest stehenden Statusleisten. */
    drawHud() {
        this.addToMap(this.healthBar);
        this.addToMap(this.coinBar);
        this.addToMap(this.bottleBar);
        if (this.endboss.activated) this.addToMap(this.endbossBar);
    }

    /**
     * Zeichnet eine Liste von Objekten.
     *
     * @param {DrawableObject[]} objects - Zu zeichnende Objekte
     */
    addObjectsToMap(objects) {
        objects.forEach((o) => this.addToMap(o));
    }

    /**
     * Zeichnet ein einzelnes Objekt auf das Canvas, ggf. gespiegelt.
     *
     * @param {DrawableObject} mo - Zu zeichnendes Objekt
     */
    addToMap(mo) {
        if (mo.otherDirection) this.flipImage(mo);
        mo.draw(this.ctx);
        if (mo.otherDirection) this.flipImageBack(mo);
    }

    /**
     * Spiegelt das Bild horizontal für die Bewegungsrichtung.
     *
     * @param {DrawableObject} mo - Zu spiegelndes Objekt
     */
    flipImage(mo) {
        this.ctx.save();
        this.ctx.translate(mo.width, 0);
        this.ctx.scale(-1, 1);
        mo.x = mo.x * -1;
    }

    /**
     * Setzt die Spiegelung zurück.
     *
     * @param {DrawableObject} mo - Gespiegeltes Objekt
     */
    flipImageBack(mo) {
        mo.x = mo.x * -1;
        this.ctx.restore();
    }
}
